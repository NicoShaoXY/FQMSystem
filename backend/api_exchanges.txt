  } else if (req.url === '/api/shenzhen' && req.method === 'GET') {
    try {
      const stocks = db.prepare(`
        SELECT s.code, s.name, sr.price, sr.change_percent, sr.volume, sr.amount
        FROM stock_realtime sr
        JOIN stocks s ON s.code = sr.code
        WHERE s.code LIKE '000%' OR s.code LIKE '001%' OR s.code LIKE '002%' OR s.code LIKE '003%'
        ORDER BY sr.change_percent DESC
        LIMIT 100
      `).all();
      
      res.end(JSON.stringify({
        success: true,
        exchange: 'shenzhen',
        count: stocks.length,
        stocks
      }));
    } catch (e) {
      res.end(JSON.stringify({ success: false, error: e.message }));
    }
  }
